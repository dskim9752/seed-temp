start_file=start_x.elf
fixup_file=fixup_x.dat

# Kernel location
kernel=zImage

# GPU memory for camera + DPI framebuffer
gpu_mem=128

# Enable Camera module
start_x=1

# CRITICAL: Disable SPI (conflicts with DPI GPIO)
dtparam=spi=off

# DPI LCD Configuration
# Force LCD, disable HDMI
ignore_lcd=0
hdmi_ignore_hotplug=1
disable_hdmi=1

# DPI GPIO assignments for RGB666 (18-bit color)
gpio=0-9=a2
gpio=12-17=a2
gpio=20-25=a2

# Enable DPI display (legacy firmware mode)
enable_dpi_lcd=1
display_default_lcd=1

# DPI timing mode
dpi_group=2
dpi_mode=87

# RGB666 output format (18-bit)
dpi_output_format=0x7F206

# 480x640 @ 60Hz timing for 2.8" DPI
hdmi_timings=480 0 26 16 10 640 0 25 10 15 0 0 0 60 0 32000000 1

# Waveshare 2.8" DPI LCD overlays
dtoverlay=waveshare-28dpi-3b-4b
dtoverlay=waveshare-28dpi-3b
dtoverlay=waveshare-touch-28dpi

# Extra transpose buffer for rotation
extra_transpose_buffer=2

# I2C for touch controller
dtparam=i2c_arm=on

# Disable audio (not used in SeedSigner)
dtparam=audio=off

# Set the bootloader delay to 0 seconds
boot_delay=0
disable_splash=1

# Security: disable wifi and bluetooth
dtoverlay=disable-wifi
dtoverlay=disable-bt
